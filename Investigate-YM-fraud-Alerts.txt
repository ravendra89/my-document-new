*
FROM prod
CROSS JOIN UNNEST(httprequest.headers) AS t(header)
WHERE header.name = 'Host'
  AND header.value like '%members.nerscinc.org'
  and from_unixtime(timestamp/1000) > now() - interval '4' hour
  --and action = 'BLOCK'
